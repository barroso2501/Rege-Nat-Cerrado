/**
 * 10d_f1_pixel_por_celula.js — Rege-Nat-Cerrado, F1, amostra P1 (passo D de 4)
 * GERADO AUTOMATICAMENTE por src/python/f1_10c_sortear_celulas.py — não edite à mão.
 * ---------------------------------------------------------------------------
 * O QUE FAZ
 *   E e P: em cada célula sorteada no passo C, escolhe UM pixel elegível de 30 m
 *     — o de menor valor de ee.Image.random(SEMENTE) entre os pixels do estrato
 *     naquele ano. ee.Image.random é determinístico (depende só da semente e da
 *     posição do pixel), então a escolha é reprodutível.
 *   N (pastagem sem evento, estrato comum): pontos aleatórios uniformes no
 *     Cerrado (fora do holdout); ficam os que caem em pixel N daquele ano.
 *   Para todos: acrescenta região (raster MapBiomas) e zona.
 *   Saída: 15 CSVs (ano × estrato E/N/P) no formato ano, estrato, lon, lat,
 *   regiao, zona, que entram no passo final f1_11_selecionar_amostra_p1.py.
 *
 *   VERSÃO 2 (2026-10-29): a v1 esgotou a memória (3 dos 5 anos). Mudanças:
 *   zona do holdout lida do asset (sem rasterizar o polígono do Cerrado);
 *   um reduceRegions por estrato em vez de 500 reduceRegion; pontos N no
 *   retângulo envolvente com rejeição; 15 tarefas menores em vez de 5.
 *
 * COMO RODAR
 *   Run → Tasks → Run nas 15 tarefas "F1_10d_candidatos_<ano>_<E|N|P>".
 *   Envie os 15 CSVs SEM ABRIR (eles têm o estrato).
 *
 * O QUE PODE QUEBRAR E COMO PERCEBER
 *   - Tarefa ainda sem memória: avise-me (próximo passo: dividir as células em lotes).

// ----------------------------- CONFIG ---------------------------------------
var C11 = 'projects/mapbiomas-public/assets/brazil/lulc/collection11/';
var cobertura = ee.Image(C11 + 'mapbiomas_brazil_collection11_coverage_v3');
var vegsec = ee.Image(C11 + 'mapbiomas_brazil_collection11_deforestation_secondary_vegetation_v5');
var vegsecCob = ee.Image(C11 + 'mapbiomas_brazil_collection11_deforestation_secondary_vegetation_by_coverage_v1');
// Zona: 0 = treino, 1 = holdout, 2 = exclusão; MASCARADA fora do Cerrado IBGE
// (script 05 v3). Usar o asset evita rasterizar o polígono do Cerrado, que
// esgotava a memória na versão anterior.
var ZONA_BRUTA = ee.Image('projects/ee-barroso2501/assets/RegeNat/F0_holdout_zona_exclusao_v1');
var REGIOES = ee.Image('users/dh-conciani/collection7/classification_regions/raster').select(0);
var CAIXA_CERRADO = ee.FeatureCollection('projects/ee-barroso2501/assets/biomas_IBGE')
  .filter(ee.Filter.eq('NM_BIOMA', 'Cerrado')).geometry().bounds();   // só o retângulo envolvente

var SEMENTE = 20261030;
var N_PONTOS_N = 15000;         // por ano, no RETÂNGULO envolvente; ~10% devem cair em pixel N
var PASTA_DRIVE = 'RegeNat_F1';
var ANOS = [2017, 2018, 2019, 2020, 2021];

// [ano, estrato, lon_centro, lat_centro, meio_lado_graus] — gerado no passo C.
var CELULAS = [
/*__CELULAS__*/
];
// ---------------------------------------------------------------------------

function banda(img, ano) { return img.select('classification_' + ano).unmask(0); }
function todosIguais(img, anos, valor) {
  var m = banda(img, anos[0]).eq(valor);
  anos.slice(1).forEach(function (a) { m = m.and(banda(img, a).eq(valor)); });
  return m;
}
// Elegível = dentro do Cerrado (zona não mascarada) e fora do holdout (zona ≠ 1).
var elegivel = ZONA_BRUTA.unmask(1).neq(1);
var ZONA = ZONA_BRUTA.unmask(-1);   // -1 = fora do Cerrado (não deve aparecer na saída)

// Máscaras dos estratos por ano.
function mascaras(t) {
  var pre15 = todosIguais(cobertura, [t - 3, t - 2, t - 1], 15);
  var pos4 = todosIguais(cobertura, [t, t + 1, t + 2, t + 3], 4);
  return {
    E: pre15.and(pos4).and(banda(vegsecCob, t).eq(515)).and(elegivel),
    P: pre15.and(pos4).and(banda(vegsec, t).eq(7)).and(elegivel),
    N: pre15.and(todosIguais(cobertura, [t, t + 1, t + 2, t + 3], 15)).and(elegivel)
  };
}

// Região e zona no ponto (uma leitura de pixel por ponto).
var regiaoZona = REGIOES.rename('regiao').addBands(ZONA.rename('zona'));

var aleatorio = ee.Image.random(SEMENTE);
var lonlat = ee.Image.pixelLonLat();

ANOS.forEach(function (t) {
  var m = mascaras(t);

  // --- E e P: um pixel por célula, numa ÚNICA chamada reduceRegions por estrato.
  ['E', 'P'].forEach(function (est) {
    var codigo = est === 'E' ? 1 : 3;
    var cel = ee.FeatureCollection(CELULAS
      .filter(function (c) { return c[0] === t && c[1] === est; })
      .map(function (c) {
        return ee.Feature(ee.Geometry.Rectangle([c[2] - c[4], c[3] - c[4], c[2] + c[4], c[3] + c[4]]));
      }));
    // min(3): menor valor de 'random' e as coordenadas desse mesmo pixel.
    var escolhido = aleatorio.addBands(lonlat).updateMask(m[est])
      .reduceRegions({collection: cel, reducer: ee.Reducer.min(3), scale: 30, tileScale: 4})
      .filter(ee.Filter.notNull(['min1']))
      .map(function (f) {
        var pt = ee.Geometry.Point([f.get('min1'), f.get('min2')]);
        return ee.Feature(pt, {ano: t, estrato: codigo, lon: f.get('min1'), lat: f.get('min2')});
      });
    var saida = regiaoZona.sampleRegions({collection: escolhido, scale: 30, geometries: false});

    Export.table.toDrive({
      collection: saida,
      description: 'F1_10d_candidatos_' + t + '_' + est,
      folder: PASTA_DRIVE,
      fileFormat: 'CSV',
      selectors: ['ano', 'estrato', 'lon', 'lat', 'regiao', 'zona']
    });
  });

  // --- N: pontos uniformes no retângulo; ficam os que caem em pixel N elegível
  // (amostragem por rejeição → uniforme dentro do estrato N).
  var pts = ee.FeatureCollection.randomPoints({region: CAIXA_CERRADO, points: N_PONTOS_N, seed: SEMENTE + t});
  var n = m.N.rename('n').addBands(regiaoZona)
    .sampleRegions({collection: pts, scale: 30, geometries: true})
    .filter(ee.Filter.eq('n', 1))
    .map(function (f) {
      var c = f.geometry().coordinates();
      return f.set({ano: t, estrato: 2, lon: c.get(0), lat: c.get(1)});
    });

  Export.table.toDrive({
    collection: n,
    description: 'F1_10d_candidatos_' + t + '_N',
    folder: PASTA_DRIVE,
    fileFormat: 'CSV',
    selectors: ['ano', 'estrato', 'lon', 'lat', 'regiao', 'zona']
  });
});
